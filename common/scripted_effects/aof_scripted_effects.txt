aof_create_planet_nation = {
	# Country Scope
	if = {
		# Only worlds with more than 500 people have a representative in the Federation.
		limit = { any_owned_colony = { NOT = { is_variable_set = aof_federation_id } count_owned_pop_amount = { count > 500 limit = { is_sapient = yes } } } }
		while = {
			limit = { any_owned_colony = { NOT = { is_variable_set = aof_federation_id } count_owned_pop_amount = { count > 500 limit = { is_sapient = yes } } } }
			federation = { save_event_target_as = current_federation }
			random_owned_colony = {
				limit = { NOT = { is_variable_set = aof_federation_id } count_owned_pop_amount = { count > 500 limit = { is_sapient = yes } } }
				save_event_target_as = current_planet
				ordered_owned_species = {
					limit = { is_sapient = yes }
					position = 0
					order_by = trigger:pop_amount
					save_event_target_as = current_species
				}
			}
			create_country = {
				name = event_target:current_planet
				type = aof_planet_nation
				species = event_target:current_species
				auto_delete = no
			}
			last_created_country = {
				# set_country_type = aof_planet_nation
				random_list = {
					1 = { change_country_flag = { icon = { category = aof_planet_state file = aof_planet_state_flag_y.dds } background = { category = backgrounds file = diagonal_stripe.dds } colors = { "yellow" "ochre_brown" "null" "null" } } }
					1 = { change_country_flag = { icon = { category = aof_planet_state file = aof_planet_state_flag.dds } background = { category = backgrounds file = diagonal_stripe.dds } colors = { "blue" "light_blue" "null" "null" } } }

				}
				# copy_ethos_and_authority = event_target:current_country
				add_modifier = { modifier = aof_world_state }
				set_country_flag = aof_planet_nation
				set_origin = origin_default
				join_alliance = { who = event_target:current_federation override_requirements = yes }
				federation = { add_cohesion = 100 }
				# copy_traditions_from = { target = event_target:current_country exceptions = {} }
				# copy_techs_from = { target = event_target:current_country } 
				# undiscovery everything
				every_galaxy_planet = {
					set_surveyed = { surveyed = no surveyor = prev }
				}
			}
			set_federation_id = { SCOPE1 = event_target:current_planet SCOPE2 = last_created_country }
		}
		aof_remove_exploration_from_world_States = yes


	}
}

aof_diplo_weight_per_pop = {
	# Country Scope
	federation = {
		set_variable = { which = diplo_weight_per_pop value = 1 }
		set_variable = { which = federation_pop_amount value = 1 }
		every_member = {
			limit = { is_aof_world_state = no }
			prev = {
				change_variable = {
					which = diplo_weight_per_pop
					value = prev.trigger:diplomacy_weight
				}
				change_variable = {
					which = federation_pop_amount
					value = prev.trigger:count_owned_pop_amount # <- but he could and he did.
				}
			}
		}
		divide_variable = { which = diplo_weight_per_pop value = federation_pop_amount }
		every_galaxy_planet = {
			limit = {
				exists = owner
				owner = { exists = federation is_in_federation_with = root }
				is_variable_set = aof_federation_id
			}
			owner = {
				federation = {
					random_member = {
						limit = { is_variable_set = aof_federation_id check_variable = { which = aof_federation_id value = PREVPREVPREV.aof_federation_id } }
						set_variable = { which = diplo_weight value = prev.diplo_weight_per_pop }
						multiply_variable = { which = diplo_weight value = PREVPREVPREV.trigger:count_owned_pop_amount }
						add_modifier = {
							modifier = aof_energy
							mult = diplo_weight
						}

					}
				}
			}
		}
	}
}

		}

			every_owned_colony = {
				clear_variable = aof_federation_id
			}

			limit = { federation_leader = { any_owned_colony = { NOT = { is_variable_set = aof_federation_id } } } }

			if = {
				limit = {
					any_member = {
						is_aof_world_state = yes
						NOT = { is_variable_set = aof_federation_id }
					}
				}
				random_member = {
					limit = { is_aof_world_state = yes }
					set_federation_id = { SCOPE1 = event_target:current_planet SCOPE2 = this }
					set_name = event_target:current_planet
				}
			}
			else = {
				event_target:current_planet = {
					ordered_owned_species = {
						limit = { is_sapient = yes }
						position = 0
						order_by = trigger:pop_amount
						save_event_target_as = current_species
					}
					create_country = {
						name = event_target:current_planet
						type = aof_planet_nation
						species = event_target:current_species
						auto_delete = no
					}
				}
				last_created_country = {
					set_country_flag = aof_planet_nation
					set_origin = origin_default
					join_alliance = { who = event_target:current_federation override_requirements = yes }

					# copy_ethos_and_authority = event_target:current_country
					# copy_traditions_from = { target = event_target:current_country exceptions = {} }
					# copy_techs_from = { target = event_target:current_country } 
				}
				set_federation_id = { SCOPE1 = event_target:current_planet SCOPE2 = last_created_country }

aof_planet_nation_happiness = {
	federation = {
		every_galaxy_planet = {
			limit = {
				exists = owner
				owner = { exists = federation is_in_federation_with = root }
				is_variable_set = aof_federation_id
			}
			owner = {
				federation = {
					random_member = {
						limit = { is_variable_set = aof_federation_id check_variable = { which = aof_federation_id value = PREVPREVPREV.aof_federation_id } }
						if = {
							limit = { PREVPREVPREV = { happiness_planet < 0.5 } }
							set_country_flag = happiness_below_5
						}
						if = {
							limit = { PREVPREVPREV = { happiness_planet >= 0.5 } }
							set_country_flag = happiness_above_5
						}
						if = {
							limit = { PREVPREVPREV = { happiness_planet > 0.8 } }
							set_country_flag = happiness_above_80
						}
					}
				}
			}
		}
	}
}

	set_country_flag = is_xenophile
}

aof_spawn_starbase_for_each_world_state = {
	every_owned_colony = {
		limit = { is_variable_set = aof_federation_id }
		save_event_target_as = current_planet
		aof_recall_matching_ID_nation = yes
		create_starbase = {
			size = starbase_starport
			owner = event_target:current_country
			module = "shipyard"
			module = "hangar_bay"
			effect = {}
		}
	}
}

aof_join_planet_state_into_war = {
	every_member = {
		limit = { is_planetless_nation = yes }
		join_war = federation.federation_leader
	}
}

aof_keep_world_states_out_of_War = {
	every_member = {
		limit = { is_aof_world_state = yes is_at_war = yes }
		federation = {
			federation_leader = {
				every_war = {
					remove_war_participant = PREVPREVPREV
				}
			}
		}
	}
}

aof_world_states_are_liked = {
	federation = {
		every_member = {
			limit = { is_aof_world_state = yes }
			every_country = {
				limit = { has_communications = prev is_ai = yes NOT = { is_same_value = prev } }
				add_opinion_modifier = { modifier = opinion_we_liberated who = prev }
			}
		}
	}
}
aof_remove_exploration_from_world_States = {
	federation = {
		every_member = {
			limit = { is_aof_world_state = yes }
			every_system = {
				set_surveyed = { surveyed = no surveyor = prev }
			}
		}
	}
}
