# our ethic checks, used for world nations without government
is_aof_xenophile = {
	OR = {
		is_xenophile = yes
		if = {
			limit = { is_scope_type = country }
			has_country_flag = is_xenophile
		}
		else = {
			owner = {
				has_country_flag = is_xenophile
			}
		}
	}
}
