@cooldown = 3600
@cohesion_loss = -20
federal_framework_low = {
	# display_level = 1
	unlocks_centralization = 4
	allow = {
		custom_tooltip = {
			NOT = {
				AND = {
					NOT = { has_federation_law = federal_framework_low }
					has_modifier = federation_law_category_federal_framework_cooldown
					has_any_federation_law_in_category = law_category_federal_framework
				}
			}
		}
	}
	on_enact = {
		add_cohesion = @cohesion_loss
		set_variable = { which = LAST_ENACTED_LAW_NUMBER value = 64 }
		if = {
			limit = { has_federation_flag = enable_federation_cooldowns }
			add_modifier = {
				modifier = federation_law_category_federal_framework_cooldown
				days = @cooldown
			}
		}
	}
	ai_weight = {
		base = 0
		
		modifier = {
			desc = federation_acceptance_generally_positive
			add = 31
		}
		modifier = {
			desc = federation_acceptance_xenophile
			add = -5
			is_aof_xenophile = yes
		}
		modifier = {
			desc = federation_acceptance_xenophobe
			add = 5
			is_aof_xenophobe = yes
		}
		modifier = {
			desc = federation_acceptance_authoritarian
			add = 5
			is_aof_authoritarian = yes
		}
		modifier = {
			desc = federation_acceptance_egalitarian
			add = -5
			is_aof_egalitarian = yes
		}
		modifier = {
			desc = federation_acceptance_spiritualist
			add = 5
			is_aof_spiritualist = yes
		}
		modifier = {
			desc = federation_acceptance_materialist
			add = -5
			is_aof_materialist = yes
		}
		modifier = {
			desc = federation_acceptance_militarist
			add = -5
			is_aof_militarist = yes
		}
		modifier = {
			desc = federation_acceptance_pacifist
			add = 5
			is_aof_pacifist = yes
		}
		#Attempt to lower centralization by a step when there is bad cohesion
		modifier = {
			desc = federation_acceptance_reduce_centralization
			add = 25
			federation = {
				federation_cohesion < -25
				has_federation_law = federal_framework_medium
			}
		}
		modifier = {
			desc = federation_acceptance_reduce_centralization
			add = 25
			federation = {
				federation_cohesion < -50
				has_federation_law = federal_framework_medium
			}
		}
		modifier = {
			desc = federation_acceptance_reduce_centralization
			add = 25
			federation = {
				federation_cohesion < -75
				has_federation_law = federal_framework_medium
			}
		}
	}
}
