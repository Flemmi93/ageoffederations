aof_union_assembly_old = {
	allow = {
		custom_tooltip = {
			NOT = {
				AND = {
					NOT = { has_federation_law = aof_union_assembly_old }
					has_modifier = federation_law_category_union_assembly_cooldown
					has_any_federation_law_in_category = law_aof_union_assembly
				}
			}
		}
		# has_federation_perk = aof_advancedd_federation_perk_3
		federation_level >= 1
	}
	cohesion_growth = 5
	on_enact = {
		if = {
			limit = { has_federation_flag = enable_federation_cooldowns }
			add_modifier = {
				modifier = federation_law_category_union_assembly_cooldown
				days = @cooldown
			}
		}
	}
	ai_weight = {
		base = 0
		modifier = {
			desc = federation_acceptance_generally_positive
			add = 26
		}

		}

			add = -5

		}
		modifier = {
			desc = federation_acceptance_authoritarian
			add = 5
			is_aof_authoritarian = yes
		}
		modifier = {
			desc = federation_acceptance_egalitarian
			add = -5
			is_aof_egalitarian = yes
		}
		modifier = {
			desc = federation_acceptance_spiritualist
			add = 5
			is_aof_spiritualist = yes
		}
		modifier = {
			desc = federation_acceptance_materialist

		}
		modifier = {
			desc = federation_acceptance_militarist

		}
		modifier = {
			desc = federation_acceptance_pacifist

			desc = federation_acceptance_poor_cohesion

		}
	}
