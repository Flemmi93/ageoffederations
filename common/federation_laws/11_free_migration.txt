@cooldown = 3600
@cohesion_loss = -20

free_migration_no = {
	potential = { NOT = { has_federation_type = planet_federation } }
	allow = {
		custom_tooltip = {
			NOT = {
				AND = {
					NOT = { has_federation_law = free_migration_no }
					has_modifier = federation_law_category_free_migration_cooldown
					has_any_federation_law_in_category = law_category_free_migration
				}
			}
		}
	}
	on_enact = {
		set_free_migration = no
		if = {
			limit = { has_federation_flag = enable_federation_cooldowns }
			add_modifier = {
				modifier = federation_law_category_free_migration_cooldown
				days = @cooldown
			}
		}
		# modded content / edit
		add_cohesion = @cohesion_loss # original value -10
		set_variable = { which = LAST_ENACTED_LAW_NUMBER value = 42 }
		###
	}

	ai_weight = {
		base = 0

		#Behaviour-specific
		modifier = {
			desc = federation_acceptance_xenophobe
			add = 75
			is_aof_xenophobe = yes
		}

		modifier = {
			desc = federation_acceptance_xenophile
			add = -50
			is_aof_xenophile = yes
		}
	}
}
