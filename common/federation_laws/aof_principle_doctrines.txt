aof_doctrines_mutualism = {
	potential = {
		OR = {
			has_federation_type = default_federation
			has_federation_type = aof_advanced_default_federation
		}
	}
	allow = {
		custom_tooltip = {
			NOT = {
				AND = {
					NOT = { has_federation_law = aof_doctrines_mutualism }
					has_modifier = federation_law_category_aof_doctrines_cooldown 
					has_any_federation_law_in_category = law_aof_principle_doctrines
				}
			}
		}
		NOT = { has_federation_law = aof_current_open_seas }
	}
	required_centralization = {
		base = @doctrine_unlock_cent
	}
	modifier = {
		planet_crime_add = -50
		pop_ethic_egalitarian_attraction_mult = 0.2
	}
	on_enact = {
		every_member = {
			aof_remove_doctrine_modifier = yes

		}
		add_cohesion = @cohesion_loss
		set_variable = { which = LAST_ENACTED_LAW_NUMBER value = 78 }
		if = {
			limit = { has_federation_flag = enable_federation_cooldowns }
			add_modifier = {
				modifier = federation_law_category_aof_doctrines_cooldown
				days = @cooldown
			}
		}
	}
	ai_weight = {

		}
