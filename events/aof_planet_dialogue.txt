			event_target:current_planet = { NOT = { has_planet_flag = aof_current_planet_security_police } }
		}
		allow = { federation = { has_federation_perk = aof_advancedd_federation_perk_2 } has_resource = { type = alloys amount >= 1000 } }
		hidden_effect = {
			country_event = {
				id = aof_planet_dip.3
				scopes = {
					from = event_target:planet_state
				}
			}
		}
		add_resource = { alloys = -1000 }
		event_target:current_planet = {
			hidden_effect = {
				set_planet_flag = aof_current_planet_security_police
			}
			create_fleet = {
				name = "event_target:current_planet"
				effect = {
					set_owner = root
					create_ship = {
						name = random
						design = "NAME_Citadel"
					}
					set_location = event_target:current_planet
					set_fleet_flag = aof_current_planet_security_police
					fleet_event = {
						id = aof_planet_dip.31
						days = 9000
						scopes = {
							from = event_target:planet_state
						}
					}
				}
			}
		}
	}
	option = {
		name = aof_planet_dip.31.b
		trigger = {
			event_target:current_planet = { has_planet_flag = aof_current_planet_security_police }
		}
		event_target:current_planet = {
			remove_planet_flag = aof_current_planet_security_police
			every_fleet_in_orbit = {
				limit = { has_fleet_flag = aof_current_planet_security_police }
