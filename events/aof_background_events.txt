	is_triggered_only = yes

		}

	}
	immediate = {
		every_system_planet = {
			save_global_event_target_as = f93_had_federation_one
			remove_planet_flag = f93_has_federation_one
		}
		FROMFROM = { federation = { every_member = { country_event = { id = f93_aof_background.131 } } } }
		every_ambient_object = {
			limit = { has_ambient_object_flag = f93_federation_one_station }
			destroy_ambient_object = this
		}
		random_galaxy_planet = {
			# Planet needs to be in federation of FROMFROM

				base = 10
				modifier = { factor = 10 is_capital = yes }
				modifier = { factor = 0.1 solar_system = { any_neighbor_system = { exists = owner starbase = { controller = { is_hostile = FROMFROM } } } } }
			}
			set_planet_flag = f93_has_federation_one
			save_global_event_target_as = f93_has_federation_one
			FROMFROM = { country_event = { id = f93_aof_market.5 } }
		}
	}
}

planet_event = {
	id = f93_aof_background.14
	hide_window = yes
	is_triggered_only = yes
	# pre_triggers = {
	# }
	trigger = {
		has_owner = yes
		has_ground_combat = no
		is_capital = no
		is_occupied_flag = no
		OR = {
			is_artificial = no
			is_ringworld = yes
		}
		# this part is to ensure only finished colonies are converted
		count_owned_pop_amount = { count >= 100 }
		has_building = yes
		
		OR = {
			owner = {
				exists = federation
				federation = { has_federation_type = planet_federation }
				count_owned_colony = {
					count > 1
					limit = { count_owned_pop_amount = { count >= 100 } }
				}
			}
			owner = {
				NOT =  { exists = federation }
				has_country_flag = activating_planet_federation
			}
		}
		NOT = {
			solar_system = {
				any_system_planet = {
					is_capital = yes
				}
			}
		}
	}
	immediate = {
		save_event_target_as = current_planet
		owner = {
			save_event_target_as = current_country
			federation = {
				save_event_target_as = current_federation
			}
		}
		create_country = {
			name = random
			type = default
			civics = random
			species = root.owner_main_species
			flag = random
			ethos = random
			origin = origin_default
			set_capital_from_species = yes
		}
		while = {
			limit = { count_owned_pop_amount = { count < 1000 } }
			event_target:current_federation = {
				random_member = {
					limit = { count_owned_pop_amount = { count > 1100 } }
					weights = {
						base = 1
						modifier = {
							factor = 0
							count_owned_pop_amount = { count = all limit = { root = { habitability = { who = prev value < 0.5 } } } }
						}
						modifier = {
							factor = 4
							count_owned_pop_amount = { count >= 100 limit = { root = { habitability = { who = prev value >= 0.5 } } } }
						}
						modifier = {
							factor = 8
							count_owned_pop_amount = { count >= 100 limit = { root = { habitability = { who = prev value >= 0.75 } } } }
						}
						modifier = {
							factor = 16
							count_owned_pop_amount = { count >= 100 limit = { root = { habitability = { who = prev value >= 1 } } } }
						}
					}
					random_owned_pop_group = {
						limit = { pop_group_size > 100 }
						weights = {
							base = 1
							modifier = {
								factor = 0.01
								root = { habitability = { who = PREV value < 0.5 } }
							}
							modifier = {
								factor = 4
								root = { habitability = { who = PREV value >= 0.5 } }
							}
							modifier = {
								factor = 8
								root = { habitability = { who = PREV value >= 0.75 } }
							}
							modifier = {
								factor = 16
								root = { habitability = { who = PREV value >= 1 } }
							}
						}
						resettle_pop_group = {
							POP_GROUP = this
							PLANET = root
							AMOUNT = 100
						}
					}
				}
			}
		}
		add_district = district_city
		add_district = district_farming
		add_district = district_mining
		add_district = district_generator
		upgrade_colony_shelter = yes
		
		if = {
			limit = { is_sector_capital = yes }
			sector = {
				every_system_within_border = {
					starbase = { set_owner = last_created_country }
				}
			}
		}
		solar_system = {
			starbase = {
				set_owner = last_created_country
				if = {
					limit = { has_starbase_size = starbase_outpost }
					set_starbase_size = starbase_starport
					set_starbase_module = {
						slot = 1
						module = shipyard
					}
				}
			}
		}
		last_created_country = {
			if = {
				limit = { event_target:current_country = { exists = federation } }
				join_alliance = {
					who = event_target:current_federation
					override_requirements = yes
				}
				federation = {
					add_cohesion = 100
				}
			}
			else = {
				join_alliance = {
					who = event_target:current_country
					override_requirements = yes
				}
				federation = {
					add_cohesion = 100
				}
			}
			copy_ethos_and_authority = event_target:current_country
			copy_traditions_from = { target = event_target:current_country exceptions = {} }
			copy_techs_from = { target = event_target:current_country }
			if = {
				limit = { event_target:current_country = { is_galactic_community_member = yes } }
				add_to_galactic_community_no_message = yes
			}
			add_resource = {
				minerals = 1200
			}
		}
		every_country = {
			limit = { has_communications = event_target:current_country }
			establish_communications_no_message = last_created_country
		}
	}
}

country_event = {
	id = f93_aof_background.16
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		exists = federation
		is_country_type = default
		federation = { has_federation_flag = aof_has_planet_nations }
	}
	immediate = {
		# If new planets are colonized and have sufficient population a new world state is established
		aof_create_planet_nation = yes
		# If new integrateable members join they are integrated. Anything that is not a subject or World State
		if = {
			limit = { federation = { is_aof_advanced_federation = yes } }
			aof_integrate_members = yes
		}
		# Updating the World States diplomatic weights. Based on the planets population - population weight is unused currently.
		if = {
			limit = { federation = { has_federation_flag = aof_uses_pop_weight } }
			federation = {
				every_member = {
					limit = { is_aof_world_state = yes }
					aof_remove_all_eco_modifications = yes
				}
			}
			aof_diplo_weight_per_pop = yes
		}
		# World States Diplo weight based on Economic Power
		else_if = {
			limit = { federation = { has_federation_flag = aof_uses_eco_weight } }
			aof_update_planet_nation_income = yes
		}
		# World States can be unhappy and this can cause issue. Happiness is tracked with population happiness of the planet
		aof_planet_nation_happiness = yes
		# World States have opinion on Votes, they are loosly based on the planets pops ethics.
		if = {
			limit = { NOT = { has_country_flag = align_world_states_cooldown } }
			aof_align_world_state_to_pop_random = yes
			aof_world_states_are_liked = yes
			set_timed_country_flag = { flag = align_world_states_cooldown days = 360 }
		}
		# ensures that World States are not war participants, as they should not be
		if = {
			limit = { any_war = { exists = this any_war_participant = { is_aof_world_state = yes } } }
			federation = { aof_keep_world_states_out_of_War = yes }
		}
